import streamlit as st
import pandas as pd

st.set_page_config(page_title="GPA Calculator", page_icon="🎓", layout="centered")
st.title("GPA Calculator")

num_subjects = st.number_input("How many subjects?", min_value=1, max_value=20, value=3, step=1)
subjects = []

for i in range(int(num_subjects)):
    st.subheader(f"Subject {i+1}")
    col1, col2, col3 = st.columns(3)
    with col1:
        name = st.text_input(f"Subject name", key=f"name_{i}", value=f"Subject {i+1}")
    with col2:
        grade_point = st.number_input(f"Grade point", key=f"grade_{i}", min_value=0.0, max_value=4.0, value=4.0, step=0.1)
    with col3:
        credit_hours = st.number_input(f"Credit hours", key=f"credit_{i}", min_value=1, max_value=6, value=3, step=1)
    subjects.append((name, grade_point, credit_hours))

if st.button("Calculate GPA", type="primary"):
    total_points = sum(g * c for _, g, c in subjects)
    total_credits = sum(c for _, _, c in subjects)
    gpa = total_points / total_credits

    df = pd.DataFrame(subjects, columns=["Subject", "Grade Point", "Credit Hours"])
    st.table(df)

    col1, col2 = st.columns(2)
    col1.metric("Total Credit Hours", total_credits)
    col2.metric("GPA", f"{gpa:.2f}")
